const REGISTRY_URLS = ['https://registry.npmjs.org/', 'http://registry.npmjs.org/'];

export default async function ({ request, target, proxy }) {
  const response = await proxy(request, target);
  if (response.status !== 200
    || !/^application\/(?:json|vnd\.npm\.install-v1\+json)(?:\s*;|$)/i.test(response.headers.get('content-type') ?? '')) {
    return response;
  }

  // The transformed representation no longer has the upstream length or validator.
  for (const header of ['content-length', 'content-encoding', 'etag', 'last-modified',
    'content-md5', 'digest', 'content-digest', 'repr-digest']) {
    response.headers.delete(header);
  }
  if (!response.body) return response;

  const replacement = JSON.stringify(`${new URL(request.url).origin}/registry.npmjs.org/`).slice(1, -1);
  let inString = false;
  let escaped = false;
  let prefix = null;
  let pending = '';
  let escape = '';
  const body = response.body
    .pipeThrough(new TextDecoderStream())
    .pipeThrough(new TransformStream({
      transform(chunk, controller) {
        let output = '';
        for (const character of chunk) {
          if (!inString) {
            output += character;
            if (character === '"') {
              inString = true;
              prefix = '';
            }
            continue;
          }

          if (prefix === null) {
            output += character;
            if (escaped) escaped = false;
            else if (character === '\\') escaped = true;
            else if (character === '"') inString = false;
            continue;
          }

          // Buffer only a possible URL prefix, including JSON escapes. Track
          // string boundaries so quoted URLs in descriptions stay untouched.
          pending += character;
          let decoded = character;
          if (escape) {
            escape += character;
            if (escape[1] === 'u' && escape.length < 6) continue;
            decoded = JSON.parse(`"${escape}"`);
            escape = '';
          } else if (character === '\\') {
            escape = character;
            continue;
          } else if (character === '"') {
            output += pending;
            pending = '';
            inString = false;
            prefix = null;
            continue;
          }

          prefix += decoded;
          if (REGISTRY_URLS.includes(prefix)) {
            output += replacement;
          } else if (REGISTRY_URLS.some(url => url.startsWith(prefix))) {
            continue;
          } else {
            output += pending;
          }
          pending = '';
          prefix = null;
        }
        controller.enqueue(output);
      },
      flush(controller) {
        controller.enqueue(pending);
      },
    }))
    .pipeThrough(new TextEncoderStream());
  return new Response(body, response);
}
